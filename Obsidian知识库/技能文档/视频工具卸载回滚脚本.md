## 按需卸载 (选择性执行)

### 卸载Manim

    $manimDir = 'E:\龙虾AI主控中心\我的AI分身\子Agent\豆包Agent\video-tools\python-libs'
    Get-ChildItem $manimDir -Directory | Where-Object { $_.Name -match 'manim|numpy|scipy|pillow|networkx|moderngl|pycairo|av-|decorator|tqdm|watchdog|rich-|click-|pygments|beautifulsoup|pydub|srt|svgelements|skia-pathops|screeninfo|isosurfaces|cloup|mapbox_earcut|audioop|manimpango|glcontext|pyglet|pyglm' } | Remove-Item -Recurse -Force
    Write-Host 'Manim及依赖已清理'

### 卸载Remotion

## 一键全量清理 (慎用)

    $vd = 'E:\龙虾AI主控中心\我的AI分身\子Agent\豆包Agent\video-tools'
    Get-ChildItem $vd -Directory | Where-Object { $_.Name -notin @('scripts','skills','knowledge-base') } | ForEach-Object {
        Write-Host "删除: $($_.Name)"
        Remove-Item $_.FullName -Recurse -Force
    }
    Write-Host '全量清理完成,核心文档已保留'

## npm缓存清理

    $cacheDir = 'E:\龙虾AI主控中心\我的AI分身\子Agent\豆包Agent\video-tools\.npm-cache'
    if (Test-Path $cacheDir) {
        Remove-Item $cacheDir -Recurse -Force
        Write-Host 'npm缓存已清理'
    }
